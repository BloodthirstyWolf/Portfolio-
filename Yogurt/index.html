document.addEventListener("DOMContentLoaded", () => { const parallaxImages = document.querySelectorAll(".parallax-img"); function updateParallax(x, y) { parallaxImages.forEach(img => { let speed = parseFloat(img.getAttribute("data-speed")); let xOffset = (window.innerWidth - x * speed) / 100; let yOffset = (window.innerHeight - y * speed) / 100; img.style.transform = `translate(${xOffset}px, ${yOffset}px)`; }); } // Определение мобильного устройства function isMobileDevice() { return /Android|iPhone|iPad|iPod/i.test(navigator.userAgent); } const isMobile = isMobileDevice(); // Обработка движения мыши (десктоп) if (!isMobile) { document.addEventListener("mousemove", (e) => { updateParallax(e.pageX, e.pageY); }); } // Обработка движения телефона (мобильные устройства) if (isMobile && window.DeviceOrientationEvent) { window.addEventListener("deviceorientation", (event) => { let x = event.gamma; // Наклон влево-вправо (-90 до 90) let y = event.beta; // Наклон вперед-назад (-180 до 180) if (x !== null && y !== null) { let mappedX = (x + 90) / 180 * window.innerWidth; let mappedY = (y + 90) / 180 * window.innerHeight; updateParallax(mappedX, mappedY); } }); } // Функция для подгонки текста под контейнер function fitTextToContainer(element) { let containerWidth = element.parentElement.clientWidth; let fontSize = 100; // Стартовый размер шрифта (px) element.style.fontSize = `${fontSize}px`; while (element.scrollWidth > containerWidth && fontSize > 10) { fontSize -= 1; element.style.fontSize = `${fontSize}px`; } } // Адаптация заголовка и подзаголовка function adjustTextSize() { document.querySelectorAll('.hero-text h1, .hero-text p').forEach(fitTextToContainer); } window.addEventListener('resize', adjustTextSize); adjustTextSize(); // Вызов при загрузке });
